167// ═══════════════════════════════════════════════════════════════════════════════
168// STM32 GPIO PORT/PIN ENCODING
169// ═══════════════════════════════════════════════════════════════════════════════
170// STM32 GPIO is addressed as (Port, Pin) — e.g., PA5 = Port A, Pin 5.
171// We encode this into hf_pin_num_t as: pin_num = (port_index * 16) + pin_index
172// where port_index: A=0, B=1, C=2, ..., K=10.
173// This allows a single hf_pin_num_t to uniquely identify any STM32 GPIO pin.
174// ─────────────────────────────────────────────────────────────────────────────
175
176namespace hf::stm32 {
177
179enum class GpioPort : hf_u8_t {
180 A = 0, B = 1, C = 2, D = 3, E = 4,
181 F = 5, G = 6, H = 7, I = 8, J = 9, K = 10
182};
183
185constexpr hf_pin_num_t EncodePin(GpioPort port, hf_u8_t pin) noexcept {
186 return static_cast<hf_pin_num_t>(static_cast<hf_u8_t>(port) * kPinsPerPort + pin);
187}
188
190constexpr hf_u8_t DecodePinPort(hf_pin_num_t pin) noexcept {
191 return static_cast<hf_u8_t>(pin / kPinsPerPort);
192}
193
195constexpr hf_u8_t DecodePinIndex(hf_pin_num_t pin) noexcept {
196 return static_cast<hf_u8_t>(pin % kPinsPerPort);
197}
198
200constexpr hf_u16_t DecodePinMask(hf_pin_num_t pin) noexcept {
201 return static_cast<hf_u16_t>(1U << DecodePinIndex(pin));
202}

548constexpr HalStatus ToHalStatus(hf_u32_t raw_status) noexcept {
549 switch (raw_status) {
550 case 0: return HalStatus::OK;
551 case 1: return HalStatus::ERROR;
552 case 2: return HalStatus::BUSY;
553 case 3: return HalStatus::TIMEOUT;
554 default: return HalStatus::ERROR;
555 }
556}
